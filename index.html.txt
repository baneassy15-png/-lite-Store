            const shippingFee = zone === 'dakar' ? settings.feeDakar : settings.feeRegion;
            const total = currentCheckoutProduct.price + shippingFee;

            document.getElementById('summary-prod-price').innerText = currentCheckoutProduct.price.toLocaleString() + ' FCFA';
            document.getElementById('summary-shipping-price').innerText = shippingFee.toLocaleString() + ' FCFA';
            document.getElementById('summary-total-price').innerText = total.toLocaleString() + ' FCFA';
        }

        function submitOrder(e) {
            e.preventDefault();
            const zone = document.querySelector('input[name="shipping-zone"]:checked').value;
            const shippingFee = zone === 'dakar' ? settings.feeDakar : settings.feeRegion;
            const total = currentCheckoutProduct.price + shippingFee;

            const newOrder = {
                id: Date.now(),
                date: new Date().toLocaleDateString('fr-FR'),
                name: document.getElementById('order-name').value,
                phone: document.getElementById('order-phone').value,
                city: zone === 'dakar' ? 'Dakar' : 'Région',
                address: document.getElementById('order-address').value,
                product: currentCheckoutProduct.name,
                total: total,
                payment: document.getElementById('order-payment').value,
                status: 'Validée'
            };

            orders.unshift(newOrder);
            saveAll();
            closeCheckoutModal();

            // Affichage de la modale de succès élégante au lieu de l'alert basique
            document.getElementById('order-success-modal').classList.remove('hidden');

            if(settings.webhookUrl) {
                fetch(settings.webhookUrl, {
                    method: 'POST',
                    mode: 'no-cors',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(newOrder)
                }).catch(err => console.log('Erreur envoi Google Sheets:', err));
            }
        }

        function openProductModal() {
            document.getElementById('prod-id').value = '';
            document.getElementById('product-form').reset();
            document.getElementById('prod-stock').value = 50;
            document.getElementById('product-modal').classList.remove('hidden');
        }
        function closeProductModal() {
            document.getElementById('product-modal').classList.add('hidden');
        }
        function saveProduct(e) {
            e.preventDefault();
            const id = document.getElementById('prod-id').value;
            const name = document.getElementById('prod-name').value;
            const category = document.getElementById('prod-category').value;
            const price = Number(document.getElementById('prod-price').value);
            const stock = Number(document.getElementById('prod-stock').value);
            const image = document.getElementById('prod-image').value;

            if(id) {
                let p = products.find(x => x.id == id);
                p.name = name; p.category = category; p.price = price; p.stock = stock; p.image = image;
            } else {
                products.push({ id: Date.now(), name, category, price, stock, image });
            }
            closeProductModal();
            saveAll();
        }
        function editProduct(id) {
            let p = products.find(x => x.id == id);
            document.getElementById('prod-id').value = p.id;
            document.getElementById('prod-name').value = p.name;
            document.getElementById('prod-category').value = p.category;
            document.getElementById('prod-price').value = p.price;
            document.getElementById('prod-stock').value = p.stock;
            document.getElementById('prod-image').value = p.image;
            document.getElementById('product-modal').classList.remove('hidden');
        }
        function deleteProduct(id) {
            products = products.filter(x => x.id != id);
            saveAll();
        }

        function addCourier() {
            const name = document.getElementById('new-courier-name').value.trim();
            if(name && !couriers.includes(name)) {
                couriers.push(name);
                document.getElementById('new-courier-name').value = '';
                saveAll();
                alert(`Livreur "${name}" ajouté avec succès !`);
            }
        }
        function removeCourier(name) {
            couriers = couriers.filter(c => c !== name);
            saveAll();
        }

        function exportOrdersToSheets() {
            if(!settings.webhookUrl) {
                alert('Veuillez d\'abord configurer votre Webhook Google Sheets dans l\'onglet Paramètres.');
                return;
            }
            orders.forEach(o => {
                fetch(settings.webhookUrl, {
                    method: 'POST',
                    mode: 'no-cors',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(o)
                }).catch(err => console.log('Erreur d\'exportation:', err));
            });
            alert('Synchronisation vers Google Sheets déclenchée avec succès !');
        }
